"""Generator for the ordered-elbow overlap-count task ``9dc70dbec7cf``.

For every nonzero color appearing exactly twice, sort its two endpoints in
row-major order, draw the canonical horizontal-then-vertical elbow, count path
multiplicity, and map counts to 3/6/9.  The generator exposes only controls that
it actually honors.  ``no_pairs`` and ``single_pair`` remain direct diagnostic
fixtures but are deliberately not advertised as lawful generator axes.
"""
from __future__ import annotations

from collections import Counter

from puzzle_generators.base import gen_ctx
from puzzle_generators.helpers.grid import full_grid

GENERATOR_ID = "9dc70dbec7cf"
VERSION = "2.1.0"
TASK_ID = GENERATOR_ID

SUMMARY = (
    "0-background grid with 2-6 distinct endpoint colors, each appearing "
    "exactly twice.  The rule connects each row-major endpoint pair with the "
    "canonical horizontal-then-vertical elbow, overlays the paths, and maps "
    "coverage counts 1/2/3+ to colors 3/6/9."
)

INVARIANTS = [
    "background is 0",
    "2-6 distinct nonzero colors each appear exactly twice",
    "all endpoint cells are distinct",
]

PALETTES = {
    "default": (1, 2, 3, 4, 5, 6, 7, 8, 9),
    "warm": (1, 2, 3, 4, 6, 7, 9),
    "cool": (1, 2, 4, 5, 6, 8, 9),
    "varied": (1, 2, 3, 4, 5, 6, 7, 8, 9),
}
PALETTE_KINDS = tuple(PALETTES)
# These are callable debugging fixtures, not advertised lawful axes.
DIAGNOSTIC_TEXTURES = ("no_pairs", "single_pair")
DEGENERATE_TEXTURES = ("all_overlap",)
HELPFUL_TEXTURES = PALETTE_KINDS

AXES = {
    "grid_h": {"type": "int", "default": "difficulty-dependent", "valid": "7..18"},
    "grid_w": {"type": "int", "default": "difficulty-dependent", "valid": "9..20"},
    "n_pairs": {"type": "int", "default": "difficulty-dependent", "valid": "2..6"},
    "palette_kind": {"type": "str", "default": "deterministic choice", "valid": "|".join(PALETTE_KINDS)},
}


def generate(seed, sample_index, *, difficulty=None, **overrides):
    texture = overrides.get("texture")
    if texture in (*DIAGNOSTIC_TEXTURES, *DEGENERATE_TEXTURES):
        return _draw_from_degenerate(texture)

    ctx = gen_ctx(
        seed=seed,
        sample_index=sample_index,
        version=VERSION,
        task_id=TASK_ID,
        difficulty=difficulty,
        overrides=overrides,
    )
    if difficulty == "easy":
        h = ctx.draw_int_strict("grid_h", 9, 11)
        w = ctx.draw_int_strict("grid_w", 9, 11)
        k_lo, k_hi = 2, 3
    elif difficulty == "hard":
        h = ctx.draw_int_strict("grid_h", 12, 16)
        w = ctx.draw_int_strict("grid_w", 14, 18)
        k_lo, k_hi = 4, 6
    else:
        h = ctx.draw_int_strict("grid_h", 9, 13)
        w = ctx.draw_int_strict("grid_w", 11, 14)
        k_lo, k_hi = 3, 5

    k = ctx.draw_int_strict("n_pairs", k_lo, k_hi)
    palette_kind = ctx.draw_choice("palette_kind", PALETTE_KINDS)
    palette = list(PALETTES[palette_kind])
    if k > len(palette):
        raise ValueError(f"palette {palette_kind!r} has fewer than {k} colors")

    rng = ctx.draw_rng("layout")
    colors = rng.sample(palette, k)
    grid = full_grid(h, w, 0)
    cells = [(r, c) for r in range(h) for c in range(w)]
    rng.shuffle(cells)
    for index, color in enumerate(colors):
        first, second = cells[2 * index], cells[2 * index + 1]
        grid[first[0]][first[1]] = color
        grid[second[0]][second[1]] = color
    return grid


def _draw_from_degenerate(name):
    grid = full_grid(10, 12, 0)
    if name == "no_pairs":
        grid[2][2] = 4
        grid[6][8] = 6
    elif name == "single_pair":
        grid[1][1] = 4
        grid[5][7] = 4
    elif name == "all_overlap":
        grid[2][1] = 4
        grid[2][8] = 4
        grid[2][4] = 6
        grid[7][4] = 6
    else:
        raise ValueError(f"unknown diagnostic texture: {name}")
    return grid


def self_check(grid, *, difficulty=None, meta=None):
    if type(grid) is not list or not grid or type(grid[0]) is not list or not grid[0]:
        return ["grid must be nonempty"]
    width = len(grid[0])
    if any(type(row) is not list or len(row) != width for row in grid):
        return ["grid must be rectangular"]
    if any(type(value) is not int or not 0 <= value <= 9 for row in grid for value in row):
        return ["cells must be exact integer colors 0..9"]

    counts = Counter(value for row in grid for value in row if value != 0)
    messages = []
    bad = {color: count for color, count in counts.items() if count != 2}
    if bad:
        messages.append(f"colors not appearing exactly twice: {bad}")
    if not 2 <= len(counts) <= 6:
        messages.append(f"expected 2..6 distinct nonzero colors, got {len(counts)}")
    return messages
